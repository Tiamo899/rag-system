# Phase 0 评测体系 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建立 golden QA 评测数据集 + 自动跑分脚本，跑出基线分数并 git 留档，为 Phase 1/2 提供前后对比依据。

**Architecture:** 评测代码放 `backend/evals/` 包：纯函数指标层（metrics）与编排层（setup_dataset 建评测库、run_eval 跑分写报告）分离。评测进程内直调 `run_rag_flow`（不走 HTTP、不碰鉴权），建库复用 TestClient + REST API（与现有集成测试同模式）。数据集与结果文件随库提交。

**Tech Stack:** Python 3.10 · pytest · FastAPI TestClient · 既有 `run_rag_flow` 服务层 · 真实 LLM/Embedding（`backend/.env`）

**Spec:** `docs/superpowers/specs/2026-09-23-thesis-roadmap-design.md`（§2 原则、§3 Phase 0 全部；Phase 1/2 由后续计划覆盖）

**对 spec 的一处落地细化（非范围变更）：** spec §3.1 要求「复制某演示政策文档为旧年份版」构造版本冲突库；演示文档均为 .docx，程序化复制+改数字脆弱。落地为两份**专门编写的 .md 政策文件**（`请假管理办法_2023/2024版.md`，关键数字 3天 vs 5天），同样实现「新旧并存、数字冲突」的测试意图，且内容确定、可维护。

## Global Constraints

- **评测先于优化**：本计划只建评测，不改 RAG 业务代码（`app/` 下文件零改动，`API.md` 不动）
- **fail-open**：LLM 裁判仅在 `llm_provider == "openai"` 时启用；mock 下跳过并在报告标注 `judge_rate: null`
- **mock 不破坏**：单测经 `tests/conftest.py` 强制 mock，指标层必须是无 I/O 纯函数；所有单测不得要求真实 API key
- **指标定义**：Hit@3 = `run_rag_flow` 返回的 `sources`（top-3）中任一 `doc_name` ∈ `expect_docs`；关键词判定 = 存在 `forbidden_keywords` 命中则败，否则 `expect_keywords` **任一**命中即胜
- **多轮基线语义**：Phase 0 每轮**独立**调 `run_rag_flow(原句, kb_id)`，不传历史——这就是现状行为，基线必须如实反映
- **依赖环境**：跑分与建库需 PG + ES 容器运行、`backend/.env` 已配真实模型；单测两者皆不需要
- **YAGNI**：不做 rerank、不做 Web UI、不做并行跑分、不做历史版本跑分对比工具（对比 = 看 git 里的 results 文件）

## Review Focus

最可能咬人的五个点，各自钉在所属任务的测试里：

1. **旧版数字出现在答案里必须判负**（即使新版数字也在）——否则 Phase 2 治理效果测不出来 → Task 1 `test_keyword_verdict_forbidden_wins`
2. **mock 模式下裁判必须跳过而非报错/算 0 分**——单测环境全 mock，一跑就崩或分数虚低 → Task 4 `test_judge_skipped_when_mock_provider`
3. **数据集字段缺失/拼错会让 runner 中途崩**——JSONL 是人工维护的 → Task 2 `test_dataset_schema`
4. **多轮基线绝不能悄悄带上历史**——那是 Phase 1 才有的行为，带了基线就失真 → Task 4 `test_multi_turn_runs_turns_independently`

```python
# backend/evals/metrics.py
"""评测指标纯函数：Hit@3 / 关键词判定 / 裁判输出解析。"""


def hit_at3(sources: list[dict], expect_docs: list[str]) -> bool:
    """top-3 来源文档命中任一期望文档。sources 元素含 doc_name。"""
    names = {s.get("doc_name") for s in sources}
    return any(d in names for d in expect_docs)


def keyword_verdict(
    answer: str, expect_keywords: list[str], forbidden_keywords: list[str]
) -> bool:
    """forbidden 任一命中 → False；否则 expect 任一命中 → True；expect 为空 → False。"""
    if any(k in answer for k in forbidden_keywords):
        return False
    if not expect_keywords:
        return False
    return any(k in answer for k in expect_keywords)


def parse_judge(text: str) -> int:
    """裁判输出 → 0/1。首字符为 '1' 判 1，否则 0。"""
    return 1 if text.strip().startswith("1") else 0
```

```python
"""数据集 schema 校验：字段齐全、文件名存在于语料、JSONL 可逐行解析。"""
import json
from pathlib import Path

DATASET = Path(__file__).resolve().parents[1] / "evals" / "dataset"
DOCS = Path(__file__).resolve().parents[1] / "evals" / "docs"
CORPUS = Path(__file__).resolve().parents[1].parent / "knowledge_base_docs_campus"


def _load(name):
    lines = (DATASET / name).read_text(encoding="utf-8").strip().splitlines()
    return [json.loads(line) for line in lines if line.strip()]


def test_single_turn_schema():
    cases = _load("single_turn.jsonl")
    assert len(cases) >= 10
    for c in cases:
        assert c["id"] and c["question"]
        assert c["expect_docs"], f"{c['id']} 缺 expect_docs"

def test_multi_turn_schema():
    cases = _load("multi_turn.jsonl")
```

```python
# backend/evals/setup_dataset.py
"""幂等建立评测库：20 份演示 docx + 2 份版本冲突 md，全部向量化。

用法（backend/ 目录下，需 PG+ES 运行、.env 已配真实模型）：
    python -m evals.setup_dataset
"""
import asyncio
import sys
import time
from pathlib import Path

from fastapi.testclient import TestClient

BACKEND_DIR = Path(__file__).resolve().parents[1]
PROJECT_DIR = BACKEND_DIR.parent
EVAL_KB_NAME = "评测库"
EVAL_CHUNK_SIZE = 500

sys.path.insert(0, str(BACKEND_DIR))


def discover_corpus(base_dir: Path | None = None) -> list[Path]:
    """演示 docx（20）+ 冲突 md（2），排序稳定、去重。"""
    root = base_dir or PROJECT_DIR
    campus = root / "knowledge_base_docs_campus"
    eval_docs = BACKEND_DIR / "evals" / "docs"
    files = sorted(campus.rglob("*.docx")) + sorted(eval_docs.glob("*.md"))
    seen: set[Path] = set()
    out = []
    for p in files:
        if p not in seen:
            seen.add(p)
            out.append(p)
    return out


def find_eval_kb(client, headers: dict) -> dict | None:
    # 列表接口无分页，直接返回全部（knowledge_bases.py: GET ""）
    r = client.get("/api/v1/knowledge-bases", headers=headers)
    for kb in r.json():
        if kb.get("name") == EVAL_KB_NAME:
            return kb
    return None


def _login(client) -> dict:
    r = client.post(
        "/api/v1/auth/login", json={"username": "admin", "password": "admin123"}
    )
    assert r.status_code == 200, f"登录失败: {r.text}"
    return {"Authorization": f"Bearer {r.json()['access_token']}"}


def _wait_completed(client, headers, doc_id, timeout=120) -> str:
    deadline = time.time() + timeout
    while time.time() < deadline:
        st = client.get(f"/api/v1/docs/{doc_id}", headers=headers).json()["status"]
        if st in ("completed", "failed"):
            return st
        time.sleep(0.5)
    return "timeout"


def _upload_one(client, headers, kb_id, path: Path) -> str:
    """上传并等向量化完成，返回状态。已存在同名且 completed 则跳过。"""
    existing = client.get(
        f"/api/v1/knowledge-bases/{kb_id}/docs",
        params={"search": path.name, "page_size": 50},
        headers=headers,
    ).json()
    for d in existing:
        if d["original_name"] == path.name and d["status"] == "completed":
            return "completed"

    content_type = (
        "application/vnd.openxmlformats-officedocument.wordprocessingml.document"
        if path.suffix == ".docx"
        else "text/markdown"
    )
    up = client.post(
        f"/api/v1/knowledge-bases/{kb_id}/docs",
        headers=headers,
        files={"file": (path.name, path.read_bytes(), content_type)},
    )
    assert up.status_code == 201, f"上传 {path.name} 失败: {up.text}"
    doc_id = up.json()["id"]
```

```python
# backend/evals/run_eval.py
"""跑分脚本：进程内调 run_rag_flow，三层指标，结果写 results/。

用法（backend/ 目录下，需 PG+ES；--tag 标记本轮，如 baseline / phase1）：
    python -m evals.run_eval --tag baseline
    python -m evals.run_eval --tag phase1 --dataset multi_turn
"""
import argparse
import asyncio
import json
from datetime import date
from pathlib import Path

from app.core.config import get_settings
from app.services.rag.langgraph.graph import run_rag_flow
from evals.metrics import hit_at3, keyword_verdict, parse_judge

DATASET_DIR = Path(__file__).resolve().parent / "dataset"
DEFAULT_RESULTS_DIR = Path(__file__).resolve().parent / "results"
DATASETS = ("single_turn", "multi_turn", "governance")

JUDGE_PROMPT = """你是严格的阅卷员。根据参考要点判断回答是否切题且覆盖要点。
只输出 1（通过）或 0（不通过），不要输出其他内容。

问题：{question}
参考要点：{keywords}
回答：{answer}
"""


def _judge_enabled() -> bool:
    return get_settings().llm_provider == "openai"


async def _judge(question: str, keywords: list[str], answer: str) -> int:
    from app.services.llm.factory import get_llm_provider

    llm = get_llm_provider()
    prompt = JUDGE_PROMPT.format(
        question=question, keywords="、".join(keywords), answer=answer[:2000]
    )
    return parse_judge(await llm.generate(prompt))


def _load_cases(name: str) -> list[dict]:
    path = DATASET_DIR / f"{name}.jsonl"
    return [
        json.loads(line)
        for line in path.read_text(encoding="utf-8").splitlines()
        if line.strip()
    ]


def _rate(values: list[bool | None]) -> float | None:
    """None（裁判跳过）全部 → None；否则按非 None 项求均值。"""
    valid = [v for v in values if v is not None]
    if not valid:
        return None
    return sum(1 for v in valid if v) / len(valid)


async def run_single_cases(
    cases: list[dict],
    kb_id: str,
    use_judge: bool,
    results_dir: Path,
    tag: str,
    dataset_name: str = "single_turn",
) -> dict:
    details = []
    hits, kws, judges = [], [], []
    for c in cases:
        plan = await run_rag_flow(c["question"], kb_id)
        answer, sources = plan["answer"], plan["sources"]
        hit = hit_at3(sources, c["expect_docs"])
        kw = keyword_verdict(
            answer, c.get("expect_keywords", []), c.get("forbidden_keywords", [])
        )
        jd = (
            await _judge(c["question"], c.get("expect_keywords", []), answer)
            if use_judge
            else None
        )
        hits.append(hit)
        kws.append(kw)
        judges.append(jd)
        details.append(
            {
                "id": c["id"],
                "hit_at3": hit,
                "keyword": kw,
                "judge": jd,
                "sources": [s.get("doc_name") for s in sources],
                "answer": answer[:300],
            }
        )
    report = {
        "tag": tag,
        "dataset": dataset_name,
        "date": date.today().isoformat(),
        "kb_id": kb_id,
        "hit_at3_rate": _rate(hits),
        "keyword_rate": _rate(kws),
        "judge_rate": _rate(judges) if use_judge else None,
        "cases": details,
    }
    _write_report(report, results_dir, tag, dataset_name)
    return report


async def run_multi_turn_cases(
    cases: list[dict],
    kb_id: str,
    use_judge: bool,
    results_dir: Path,
    tag: str,
) -> dict:
    """Phase 0 基线：每轮独立调用，不携带历史。Phase 1 计划将改为传历史。"""
    details = []
    hits, kws, judges = [], [], []
    for c in cases:
        turn_rows = []
        for t in c["turns"]:
            plan = await run_rag_flow(t["q"], kb_id)
            answer, sources = plan["answer"], plan["sources"]
            hit = hit_at3(sources, t.get("expect_docs", []))
            kw = keyword_verdict(
                answer, t.get("expect_keywords", []), t.get("forbidden_keywords", [])
            )
            jd = (
                await _judge(t["q"], t.get("expect_keywords", []) or ["要点"], answer)
                if use_judge and t.get("expect_keywords")
                else None
            )
            hits.append(hit)
            kws.append(kw)
            judges.append(jd)
            turn_rows.append(
                {
                    "q": t["q"],
                    "hit_at3": hit,
                    "keyword": kw,
                    "judge": jd,
                    "sources": [s.get("doc_name") for s in sources],
                    "answer": answer[:300],
                }
            )
        details.append({"id": c["id"], "turns": turn_rows})
    report = {
        "tag": tag,
        "dataset": "multi_turn",
        "date": date.today().isoformat(),
        "kb_id": kb_id,
        "hit_at3_rate": _rate(hits),
        "keyword_rate": _rate(kws),
        "judge_rate": _rate(judges) if use_judge else None,
        "cases": details,
    }
    _write_report(report, results_dir, tag, "multi_turn")
    return report


def _write_report(report: dict, results_dir: Path, tag: str, name: str) -> None:
    results_dir.mkdir(parents=True, exist_ok=True)
    stem = f"{report['date']}-{tag}-{name}"
    (results_dir / f"{stem}.json").write_text(
        json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    judge_line = (
        f"{report['judge_rate']:.2%}"
        if report["judge_rate"] is not None
        else "跳过（mock provider）"
    )
    lines = [
        f"# 评测报告 · {stem}",
        "",
        f"- tag: `{tag}`  dataset: `{name}`  date: {report['date']}",
        f"- kb_id: `{report['kb_id']}`",
        f"- Hit@3: **{report['hit_at3_rate']:.2%}**",
        f"- 关键词: **{report['keyword_rate']:.2%}**",
        f"- 裁判: **{judge_line}**",
        "",
        "| id | Hit@3 | 关键词 | 裁判 | 来源 |",
        "|----|-------|--------|------|------|",
    ]
    for c in report["cases"]:
        if "turns" in c:
            for i, t in enumerate(c["turns"]):
                lines.append(
                    f"| {c['id']}-t{i + 1} | {t['hit_at3']} | {t['keyword']} "
                    f"| {t['judge']} | {', '.join(x or '' for x in t['sources'])} |"
                )
        else:
            lines.append(
                f"| {c['id']} | {c['hit_at3']} | {c['keyword']} "
                f"| {c['judge']} | {', '.join(x or '' for x in c['sources'])} |"
            )
    (results_dir / f"{stem}.md").write_text("\n".join(lines), encoding="utf-8")


async def _amain(args: argparse.Namespace) -> None:
    from evals.setup_dataset import EVAL_KB_NAME, find_eval_kb
    from fastapi.testclient import TestClient
    from app.main import app

    with TestClient(app) as client:
        r = client.post(
            "/api/v1/auth/login",
            json={"username": "admin", "password": "admin123"},
        )
        headers = {"Authorization": f"Bearer {r.json()['access_token']}"}
        kb = find_eval_kb(client, headers)
    assert kb is not None, "评测库不存在，先运行 python -m evals.setup_dataset"

    use_judge = _judge_enabled()
    results_dir = Path(args.results_dir)
    targets = DATASETS if args.dataset == "all" else (args.dataset,)
    print(f"kb={kb['id']} judge={'on' if use_judge else 'off'} datasets={targets}")
    for name in targets:
        cases = _load_cases(name)
        if name == "multi_turn":
            report = await run_multi_turn_cases(
                cases, kb["id"], use_judge, results_dir, args.tag
            )
        else:
            report = await run_single_cases(
                cases, kb["id"], use_judge, results_dir, args.tag,
                dataset_name=name,
            )
        print(
            f"[{name}] Hit@3={report['hit_at3_rate']:.2%} "
            f"keyword={report['keyword_rate']:.2%} "
            f"judge={report['judge_rate']}"
        )


def main() -> None:
    parser = argparse.ArgumentParser(description="RAG 评测跑分")
    parser.add_argument("--tag", required=True, help="baseline/phase1/phase2…")
    parser.add_argument("--dataset", default="all",
                        choices=("all",) + DATASETS)
    parser.add_argument("--results-dir", default=str(DEFAULT_RESULTS_DIR))
    args = parser.parse_args()
    asyncio.run(_amain(args))


if __name__ == "__main__":
    main()
```
